   ❌ ANTES:
   ┌────────────────────────────────────────────────────────────────────────┐
   │ class _PostCard extends StatelessWidget {                              │
   │   final Post post;                                                     │

   │   final Post post;                                                     │
   │   final String? myId;                                                  │
   │   const _PostCard({required this.post, this.myId});                    │
   │   ...                                                                  │
   │   if (post.authorId != myId)  // ✅ myId ahora disponible              │
